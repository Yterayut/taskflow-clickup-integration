// Fix checkAuthentication function to use existing API async function checkAuthentication() { try { // Use dashboard analytics API instead of non-existent verify API const response = await fetch('/api/v2/dashboard/analytics', { method: 'GET', credentials: 'include' }); if (response.ok) { const data = await response.json(); if (data.success) { // Get user info from localStorage (saved during login) const storedUser = localStorage.getItem('taskflow-user'); if (storedUser) { try { const user = JSON.parse(storedUser); return { success: true, user: user }; } catch (e) { console.error('Error parsing stored user:', e); } } // Default user if no stored data return { success: true, user: { name: 'Teerayut Yeerahem', email: 'yterayut@gmail.com', role: 'admin' } }; } } return { success: false }; } catch (error) { console.error('Authentication check error:', error); return { success: false }; } }
